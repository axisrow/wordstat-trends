# Обход для transitive-зависимости browser-use у wordstat-cli.
# Нужен ТОЛЬКО для `uv pip install` — обычный `pip install` эту проблему не
# имеет вовсе (см. ниже).
#
# wordstat-cli/pyproject.toml объявляет:
#   [tool.uv.sources]
#   browser-use = { path = "../browser-use", editable = true }
# — путь относительно репозитория wordstat-cli. Вне его дерева (в т.ч. здесь,
# при установке wordstat как git-зависимости) такого каталога нет, и
# `uv pip install` падает на сборке browser-use. `[tool.uv.sources]` в этом
# pyproject.toml не помогает: он переопределяет источники только для прямых
# зависимостей wordstat-trends, а не транзитивных зависимостей внутри пакета
# `wordstat`. `pip` же вообще не читает `[tool.uv.sources]` (это расширение
# uv) и разрешает browser-use с PyPI сам, без всякого обхода — проверено:
# `python -m pip install -e ".[dev]"` в чистом venv отрабатывает без
# --override. См. docs/STACK.md → «browser-use: обходной путь для установки».
#
# Использование (только под uv):
#   uv pip install -e ".[dev]" --override install-overrides.txt
browser-use==0.13.8
